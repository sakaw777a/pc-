Sample Query — Fetch Compatible Motherboards for Selected CPU
SELECT c.id, c.name, c.price_usd, ms.chipset, ms.ram_type, ms.pcie_version
FROM components c
JOIN motherboard_specs ms ON ms.component_id = c.id
WHERE c.type = 'motherboard'
  AND ms.socket = (
    SELECT cs.socket FROM cpu_specs cs
    WHERE cs.component_id = :selected_cpu_id
  )
ORDER BY c.price_usd ASC;
